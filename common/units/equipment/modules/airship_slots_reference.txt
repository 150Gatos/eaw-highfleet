# Add module_slots and default_modules to your existing archetypes.
# This is a snippet showing what to add — merge into your existing files.

# ── LIGHT AIRSHIP ─────────────────────────────────────────────────────────────

# In light_highfleet.txt, add to light_highfleet_equipment archetype:
#
#   module_slots = {
#       main_armament_slot = {
#           required = yes
#           allowed_module_categories = { airship_main_gun_slot }
#       }
#       secondary_slot = {
#           required = no
#           allowed_module_categories = { airship_secondary_slot }
#       }
#       engine_slot = {
#           required = yes
#           allowed_module_categories = { airship_engine_slot }
#       }
#       armor_slot = {
#           required = yes
#           allowed_module_categories = { airship_armor_slot }
#       }
#       lift_slot = {
#           required = yes
#           allowed_module_categories = { airship_lift_slot }
#       }
#       navigation_slot = {
#           required = no
#           allowed_module_categories = { airship_navigation_slot }
#       }
#       aa_slot = {
#           required = no
#           allowed_module_categories = { airship_aa_slot }
#       }
#       crew_slot = {
#           required = yes
#           allowed_module_categories = { airship_crew_slot }
#       }
#       special_slot = {
#           required = no
#           allowed_module_categories = { airship_special_slot }
#       }
#       auxiliary_slot = {
#           required = no
#           allowed_module_categories = { airship_auxiliary_slot }
#       }
#   }
