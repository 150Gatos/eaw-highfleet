# ── Module Categories ─────────────────────────────────────────────────────────
# These are the slot types. Each module must reference one of these as its category.
# Each slot on equipment must reference one or more of these in allowed_module_categories.

equipment_module_categories = {

    airship_main_gun_slot       # primary weapons battery
    airship_secondary_slot      # secondary/broadside armament
    airship_engine_slot         # propulsion system
    airship_armor_slot          # hull plating / protection
    airship_lift_slot           # buoyancy / lifting system
    airship_navigation_slot     # sensors, rangefinders, comms
    airship_aa_slot             # anti-air mounts
    airship_crew_slot           # crew quarters, command facilities
    airship_special_slot        # unique special equipment
    airship_auxiliary_slot      # miscellaneous auxiliary systems
}
