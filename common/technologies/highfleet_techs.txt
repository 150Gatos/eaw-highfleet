## The tech tree works by having two identical tech trees overlapping each other, one for players with the NSB DLC and another for players without it.
## Each tech tree is only visible depending on whether the player has the NSB DLC. This is to maintain compatibility with players who do not have it.
## Both tech trees should give approximately the same stats to avoid different experiences.
## The tech tree without NSB will have longer research times to balance the lack of XP requirements (It should be feasable to tech rush).
##
## The following is an example with explanation of how a tech is structured
##
## Standard of the name of a tech for non NSB follows this structure type_highfleet_model (the model starts at 0).
## light_highfleet_0 = { 
## 'on_research_complete' encompases the reward the player will have.
##     on_research_complete = {
## 'enable_equipments' is a reward of the on_research_complete, it unlocks an equipment, on_research_complete can give any effect also (See wiki for all).
##         enable_equipments = { light_highfleet_equipment_0 }
##     }
## 'enable_subunits' is a reward of the on_research_complete, it unlocks the use of troop types (This includes both battalions and support companies).
##     enable_subunits = {
## This is a 'battalion', AKA a troop you put in a division, it follows this structure type_highfleet_fleet/ship (fleet if its more than one, ship if its an individual entity).
##         light_highfleet_fleet     
## This is a 'detachment', AKA a support company you put in a division, it follows this structure type_highfleet_detachment (A detachment uses half of a battalion).
##         light_highfleet_detachment
##     }
## 'visible' makes it visible for players with the condition inside it, in this case players without the DLC NSB (No Step Back) since it also has NOT block inside. It can also accept other requierements
##     visible = { NOT = { has_dlc = "No Step Back" } }
## 'allow' makes it researchable for players with the condition inside it, in this case players without the DLC NSB (No Step Back) since it also has NOT block inside. It can also accept other requierements
##     allow = { NOT = { has_dlc = "No Step Back" } }
## 'path' is used to designate what tech it has next (one tech can lead to multiple), it also makes the arrow in the gui.
##     path = { leads_to_tech = light_highfleet_1 }
## 'sub_technologies' are technologies that appear on the same box below of the parent technology (They follow the same structure as a normal tech, but dont have a path).
##     sub_technologies = { light_highfleet_a_0 light_highfleet_b_0 light_highfleet_c_0 }
## 'research_cost' allows control over the formula the game uses to calculate how long it will take. (days = research_cost × 212 / research_speed_factor).
##     research_cost = 1.2
## 'start_year' designates when the tech will start being ahead of time, when a player researches the tech before the date it suffers an increase in research time (+236 each 1.0 years ahead of time).
##     start_year = 1005
##     folder = { name = highfleet_techs position = { x = @1005 y = @light } }
##     categories = {
##         highfleet_category
##     }
##     special_project_specialization = { specialization_land }
##     ai_will_do = {
##         factor = 1
##         modifier = { factor = 2 is_major = yes }
##     }
## }
